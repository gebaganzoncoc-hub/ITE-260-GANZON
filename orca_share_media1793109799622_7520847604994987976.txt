# ---------- VALIDATION FUNCTIONS ----------

def valid_student_id(student_id):
    parts = student_id.split("-")

    return (
        len(parts) == 3
        and parts[0].isdigit() and len(parts[0]) == 2
        and parts[1].isdigit() and len(parts[1]) == 4
        and parts[2].isdigit() and len(parts[2]) == 6
    )


def valid_name(name):
    return bool(name.strip()) and all(
        character.isalpha() or character in " ., "
        for character in name
    )


def valid_contact(contact):
    return contact.isdigit() and len(contact) == 11


def valid_username(username):
    return username.isalnum()


def valid_password(password):
    return (
        8 <= len(password) <= 15
        and password.isalnum()
        and any(character.isalpha() for character in password)
        and any(character.isdigit() for character in password)
    )


def select_option(title, options):
    while True:
        print("\n" + title)

        for number, option in enumerate(options, start=1):
            print(f"{number}. {option}")

        choice = input("Enter choice: ").strip()

        if choice.isdigit():
            choice = int(choice)

            if 1 <= choice <= len(options):
                return options[choice - 1]

        print("Invalid choice. Please try again.")


def find_student(student_id):
    for student in students:
        if student[0] == student_id:
            return student

    return None


def input_student_id():
    while True:
        student_id = input(
            "Enter Student ID (00-0000-000000): "
        ).strip()

        if not valid_student_id(student_id):
            print("Invalid format. Use 00-0000-000000.")
        elif find_student(student_id):
            print("Student ID already exists.")
        else:
            return student_id


def input_name():
    while True:
        name = input("Enter Full Name: ").strip()

        if valid_name(name):
            return name

        print("Use letters, spaces, commas, or periods only.")


def input_contact():
    while True:
        contact = input(
            "Enter Contact Number (11 digits): "
        ).strip()

        if valid_contact(contact):
            return contact

        print("Contact number must contain exactly 11 digits.")


def input_existing_student_id():
    while True:
        student_id = input(
            "Enter Student ID (or 0 to return): "
        ).strip()

        if student_id == "0":
            return None

        if not valid_student_id(student_id):
            print("Invalid format. Use 00-0000-000000.")
            continue

        student = find_student(student_id)

        if student is None:
            print("Student record not found. Try again.")
            continue

        return student


# ---------- LOGIN ----------

def login():
    while True:
        print("\n" + "=" * 40)
        print(" STUDENT INFORMATION SYSTEM")
        print("=" * 40)

        username = input("Username: ").strip()

        if not valid_username(username):
            print("Username must contain letters and numbers only.")
            continue

        if username not in accounts:
            print("Account not found. Please try again.")
            continue

        account = accounts[username]

        while True:
            password = input("Password: ").strip()

            if not valid_password(password):
                print("Password must be 8-15 characters.")
                print("Include at least one letter and one number.")
                continue

            if password != account["password"]:
                print("Incorrect password. Try again.")
                continue

            break

        role = account["role"]

        print("\n" + "=" * 40)
        print(" LOGIN SUCCESSFUL")
        print("Welcome,", role + "!")
        print("=" * 40)

        if role == "Instructor":
            section = section_assignment()
            instructor_menu(section)

        elif role == "Admin":
            admin_menu()

        elif role == "Student":
            student_menu(account["student_id"])

# ---------- UPDATE STUDENT ----------

def update_student():
    print("\n" + "=" * 40)
    print(" UPDATE STUDENT RECORD")
    print("=" * 40)

    student = input_existing_student_id()

    if student is None:
        return

    while True:
        print("\n1. Student ID")
        print("2. Full Name")
        print("3. Course")
        print("4. Year Level")
        print("5. Student Status")
        print("6. Contact Number")
        print("7. Section")
        print("8. Finish Updating")

        choice = input("Enter choice: ").strip()

        if choice == "1":
            while True:
                new_id = input(
                    "Enter new Student ID (00-0000-000000): "
                ).strip()

                if not valid_student_id(new_id):
                    print("Invalid Student ID format.")
                    continue

                if find_student(new_id):
                    print("Student ID already exists.")
                    continue

                old_id = student[0]
                student[0] = new_id

                for account in accounts.values():
                    if account.get("student_id") == old_id:
                        account["student_id"] = new_id

                if old_id in attendance:
                    attendance[new_id] = attendance.pop(old_id)

                print("Student ID updated successfully.")
                break

        elif choice == "2":
            student[1] = input_name()
            print("Name updated successfully.")

        elif choice == "3":
            student[2] = select_option("Select Course:", courses)
            print("Course updated successfully.")

        elif choice == "4":
            student[3] = select_option(
                "Select Year Level:", year_levels
            )
            print("Year level updated successfully.")

        elif choice == "5":
            student[4] = select_option(
                "Select Student Status:", statuses
            )
            print("Student status updated successfully.")

        elif choice == "6":
            student[5] = input_contact()
            print("Contact number updated successfully.")

        elif choice == "7":
            student[6] = select_option(
                "Select New Section:", sections
            )
            print("Section updated successfully.")

        elif choice == "8":
            print("Finished updating student record.")
            return

        else:
            print("Invalid choice. Please select 1-8.")


# ---------- DELETE STUDENT ----------

def delete_student():
    print("\n" + "=" * 40)
    print(" DELETE STUDENT RECORD")
    print("=" * 40)

    student = input_existing_student_id()

    if student is None:
        return

    print("\nStudent ID:", student[0])
    print("Full Name:", student[1])
    print("Section:", student[6])

    while True:
        confirm = input(
            "Delete this record? (Y/N): "
        ).strip().upper()

        if confirm == "Y":
            student_id = student[0]
            students.remove(student)

            linked_accounts = [
                username for username, account in accounts.items()
                if account.get("student_id") == student_id
            ]

            for username in linked_accounts:
                del accounts[username]

            attendance.pop(student_id, None)

            print("Student record deleted successfully.")
            return

        elif confirm == "N":
            print("Deletion cancelled.")
            return

        else:
            print("Please enter Y or N.")

def record_attendance(section=None):
    print("\n" + "=" * 40)
    print(" RECORD STUDENT ATTENDANCE")
    print("=" * 40)

    while True:
        student_id = input(
            "Enter Student ID (or 0 to return): "
        ).strip()

        if student_id == "0":
            return

        if not valid_student_id(student_id):
            print("Invalid format. Use 00-0000-000000.")
            continue

        student = find_student(student_id)

        if student is None:
            print("Student not found. Try again.")
            continue

        if section is not None and student[6] != section:
            print("Student is not in your assigned section.")
            continue

        break

    print("Student:", student[1])
    print("Section:", student[6])
    print("Date:", date.today().isoformat())

    attendance_status = select_option(
        "Select Attendance Status:",
        ["Present", "Absent", "Late", "Excused"]
    )

    attendance_date = input(
        "Enter attendance date (YYYY-MM-DD), "
        "or press Enter for today: "
    ).strip()

    if not attendance_date:
        attendance_date = date.today().isoformat()

    try:
        date.fromisoformat(attendance_date)
    except ValueError:
        print("Invalid date. Use YYYY-MM-DD.")
        return

    records = attendance.setdefault(student_id, {})

    if attendance_date in records:
        confirm = input(
            "Attendance already exists. Replace it? (Y/N): "
        ).strip().upper()

        if confirm != "Y":
            print("Attendance update cancelled.")
            return

    records[attendance_date] = attendance_status
    print("Attendance recorded successfully.")
    print("Student:", student[1])
    print("Date:", attendance_date)
    print("Status:", attendance_status)

def view_my_attendance(student_id):

    print(" MY ATTENDANCE RECORDS")
    print("=" * 40)

    student = find_student(student_id)

    if student is None:
        print("Student record not found.")
        return

    print("Student:", student[1])
    records = attendance.get(student_id, {})

    if not records:
        print("No attendance records yet.")
        return

    for attendance_date in sorted(records):
        print(attendance_date, "-", records[attendance_date])


# ---------- VIEW ATTENDANCE REPORT ----------

def view_all_attendance(section=None):
    print("\n" + "=" * 40)
    print(" ATTENDANCE REPORT")
    print("=" * 40)

    found = False

    for student in students:
        if section is not None and student[6] != section:
            continue

        records = attendance.get(student[0], {})

        for attendance_date, status in sorted(records.items()):
            print(
                student[1], "|",
                student[0], "|",
                student[6], "|",
                attendance_date, "|",
                status
            )
            found = True

    if not found:
        print("No attendance records found.")
